Implement an RNN to generate text based on a corpus of text.

import tensorflow as tf import numpy as np import matplotlib.pyplot as plt # Text corpus text = """machine learning is a branch of artificial intelligence. deep learning uses neural networks. recurrent neural networks process sequential data. machine learning helps computers learn from data. deep learning is useful for image and text processing.""" # Tokenization tokenizer = tf.keras.preprocessing.text.Tokenizer() tokenizer.fit_on_texts([text]) seq = tokenizer.texts_to_sequences([text])[0] # Create input and output sequences X, y = [], [] for i in range(2, len(seq)): X.append(seq[i-2:i]) y.append(seq[i]) X, y = np.array(X), np.array(y) vocab = len(tokenizer.word_index) + 1 # Build Simple RNN model model = tf.keras.Sequential([ tf.keras.layers.Embedding(vocab, 16), tf.keras.layers.SimpleRNN(32), tf.keras.layers.Dense(vocab, activation='softmax') ]) # Compile model model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy']) # Train model history = model.fit(X, y, epochs=100, verbose=0) # Plot accuracy and loss plt.figure(figsize=(10, 4)) plt.subplot(1, 2, 1) plt.plot(history.history['accuracy']) plt.title('Training Accuracy') plt.xlabel('Epochs') plt.ylabel('Accuracy') plt.subplot(1, 2, 2) plt.plot(history.history['loss']) plt.title('Training Loss') plt.xlabel('Epochs') plt.ylabel('Loss') plt.tight_layout() plt.show() # Generate text seed = "machine learning" for _ in range(10): tokens = tokenizer.texts_to_sequences([seed])[0][-2:] tokens = tf.keras.preprocessing.sequence.pad_sequences( [tokens], maxlen=2) pred = model.predict(tokens, verbose=0).argmax() seed += " " + tokenizer.index_word.get(pred, "") print("Generated Text:\n", seed)
