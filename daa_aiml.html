0/1 Knapsack (DP)

#0/1 Knapsack using dynamic programming def knapsack(weight, value, capacity): n = len(value) dp = [[0]*(capacity+1) for _ in range(n+1)] for i in range(1, n+1): for w in range(1, capacity+1): if weight[i-1]<=w: dp[i][w] = max(dp[i-1][w], value[i-1] + dp[i-1][w - weight[i-1]]) else: dp[i][w]=dp[i-1][w] return dp[n][capacity] weights = [1,2,3] values = [6,10,12] capacity = 5 max_value = knapsack(weights, values, capacity) print("Max Value of 0/1 Knapsack using DP: ", max_value)

0/1 Knapsack (BB)

def knapsack_bb(weights, values, capacity): n = len(values) # Sort by value/weight ratio descending idx = sorted(range(n), key=lambda i: values[i]/weights[i], reverse=True) w = [weights[i] for i in idx] v = [values[i] for i in idx] max_profit = 0 def bound(i, profit, weight): """Calculate upper bound using fractional knapsack""" while i < n and weight + w[i] <= capacity: profit += v[i] weight += w[i] i += 1 # Add fraction of next item if possible if i < n: profit += (capacity - weight) * v[i] / w[i] return profit def dfs(i, profit, weight): nonlocal max_profit # Prune if bound is not promising if bound(i, profit, weight) <= max_profit: return # Update max profit if profit > max_profit: max_profit = profit # Base case if i >= n: return # Branch 1: Include item i if weight + w[i] <= capacity: dfs(i + 1, profit + v[i], weight + w[i]) # Branch 2: Exclude item i dfs(i + 1, profit, weight) dfs(0, 0, 0) return max_profit # Example values = [6, 10, 12] weights = [1, 2, 3] capacity = 5 print("Maximum value:", knapsack_bb(weights, values, capacity))

#huffman coding using Greedy Method import heapq from collections import Counter class Node: def __init__(self, char=None, freq=0, left=None, right=None): self.char, self.freq = char, freq self.left, self.right = left, right def __lt__(self, other): return self.freq < other.freq # Build tree def build_tree(text): heap = [Node(c, f) for c, f in Counter(text).items()] heapq.heapify(heap) while len(heap) > 1: a, b = heapq.heappop(heap), heapq.heappop(heap) heapq.heappush(heap, Node(None, a.freq + b.freq, a, b)) return heap[0] # Generate codes def get_codes(node, code="", codes={}): if node: if node.char: codes[node.char] = code get_codes(node.left, code + "0", codes) get_codes(node.right, code + "1", codes) return codes # Encode & Decode def encode(text): root = build_tree(text) codes = get_codes(root, "", {}) return ''.join(codes[c] for c in text), codes, root def decode(encoded, root): res, node = "", root for bit in encoded: node = node.left if bit == "0" else node.right if node.char: res += node.char node = root return res # Example text = "huffman coding is greedy" enc, codes, tree = encode(text) print("Codes:", codes) print("Encoded:", enc) print("Decoded:", decode(enc, tree))
